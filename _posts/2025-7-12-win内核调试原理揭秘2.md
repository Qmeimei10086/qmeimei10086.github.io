  PETHREAD LastThread; // [rsp+38h] [rbp-60h] BYREF

  FirstThread = 0;
  memset(&apc, 0, sizeof(apc));
  LastThread = 0;
  result = DbgkpPostFakeThreadMessages(Process, DebugObject, 0, &FirstThread, &LastThread);
  if ( result >= 0 )
  {

  return result;
}
